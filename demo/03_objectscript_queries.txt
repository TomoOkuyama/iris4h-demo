// ============================================================
// デモ シナリオ2: SQL + ObjectScript 同時クエリ
// IRIS ターミナルで実行（ネームスペース: FHIRSERVER）
//
// 接続方法:
//   docker exec -it iris4h iris session IRIS -U FHIRSERVER
// ============================================================

// --- 方法1: ObjectScript から SQL を実行 ---
Set stmt = ##class(%SQL.Statement).%New()
Do stmt.%Prepare("SELECT Key, subject_Reference, date_StartRaw FROM HSFHIR_X0001_S.Observation")
Set rs = stmt.%Execute()
While rs.%Next() { Write rs.Key, " | ", rs.%Get("subject_Reference"), " | ", rs.%Get("date_StartRaw"), ! }

// --- 方法2: FHIR Server API でリソースをオブジェクトとして取得 ---
// （FHIRリソースのJSON全体にアクセスできる）
Set appKey = "/csp/healthshare/fhirserver/fhir/r4"
Set strategy = ##class(HS.FHIRServer.API.InteractionsStrategy).GetStrategyForEndpoint(appKey)
Set interactions = strategy.NewInteractionsInstance()
Set json = interactions.Read("Observation", "4")
Write "resourceType: ", json.resourceType, !
Write "O2Sat: ", json.valueQuantity.value, " ", json.valueQuantity.unit, !
Write "患者参照: ", json.subject.reference, !
Write "測定日時: ", json.effectiveDateTime, !

// --- 方法3: 同じデータをSQLでも確認 ---
Set stmt = ##class(%SQL.Statement).%New()
Do stmt.%Prepare("SELECT Key, subject_Reference FROM HSFHIR_X0001_S.Observation WHERE Key = 'Observation/4'")
Set rs = stmt.%Execute()
If rs.%Next() { Write "SQL結果: ", rs.Key, " → ", rs.%Get("subject_Reference"), ! }

// --- 方法4: グローバル変数（データの実体）を直接確認 ---
// FHIRリソースは最終的にグローバル変数に格納されている
ZWrite ^HSFHIR.X0001.S.ObservationD(1)
